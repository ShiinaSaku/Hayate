import { useState } from 'react';

export function InstallCard({ title, icon, command }) {
  const [copied, setCopied] = useState(false);

  const handleCopy = () => {
    navigator.clipboard.writeText(command);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div style={{
      border: '1px solid var(--rp-c-divider)',
      borderRadius: '12px',
      padding: '24px',
      background: 'var(--rp-c-bg-soft)',

        {icon}
        {title}
      </h3>
      <p style={{ color: 'var(--rp-c-text-2)', fontSize: '0.9rem', marginBottom: '16px' }}>
        Execute this command in your terminal:
      </p>
      <div style={{ display: 'flex', alignItems: 'center', gap: '10px', background: 'var(--rp-c-bg-mute)', border: '1px solid var(--rp-c-divider-light)', padding: '12px 16px', borderRadius: '8px', position: 'relative' }}>
        <code style={{ fontSize: '0.85rem', color: 'var(--rp-c-text-1)', fontFamily: 'var(--rp-c-font-mono)', overflowX: 'auto', whiteSpace: 'nowrap', flex: 1, paddingRight: '40px' }}>
          {command}
        </code>
        <button
          onClick={handleCopy}
          style={{
            position: 'absolute',
            right: '12px',
            top: '50%',
            transform: 'translateY(-50%)',
            background: 'var(--rp-c-bg)',
            border: '1px solid var(--rp-c-divider)',
            borderRadius: '6px',
            padding: '6px',
            cursor: 'pointer',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            color: 'var(--rp-c-text-2)',
            transition: 'all 0.2s'
          }}
          title="Copy command"
        >
          {copied ? (
            <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="#22c55e" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><polyline points="20 6 9 17 4 12"/></svg>
          ) : (
            <svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><rect x="9" y="9" width="13" height="13" rx="2" ry="2"/><path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1"/></svg>
          )}
        </button>
      </div>
    </div>
  );
}
